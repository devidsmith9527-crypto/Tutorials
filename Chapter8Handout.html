using Microsoft.AspNetCore.Mvc;
using SchoolApp.Data;
using SchoolApp.Models;

namespace SchoolApp.Controllers
{
    public class StudentController : Controller
    {
        private readonly SchoolDbContext _context;
        public StudentController(SchoolDbContext context) { _context = context; }

        // ១. GET: /Student/Create (បង្ហាញ Form ទទេ)
        [HttpGet]
        public IActionResult Create()
        {
            return View();
        }

        // ២. POST: /Student/Create (ទទួលទិន្នន័យមក Save)
        [HttpPost]
        [ValidateAntiForgeryToken] // ការពារការវាយប្រហារ CSRF
        public async Task<IActionResult> Create(Student model)
        {
            // ឆែកមើល Validation តាម Model Rules
            if (!ModelState.IsValid)
            {
                return View(model); // បើខុស បង្ហាញ Form វិញអមជាមួយ Error
            }

            await _context.Students.AddAsync(model);
            await _context.SaveChangesAsync();

            TempData["Success"] = "រក្សាទុកទិន្នន័យសិស្សជោគជ័យ!";
            return RedirectToAction(nameof(Index)); // 302 Redirect ទៅទំព័របញ្ជី
        }
    }
}

Controllers/StudentApiController.cs
using Microsoft.AspNetCore.Mvc;
using SchoolApp.Data;
using SchoolApp.Models;

namespace SchoolApp.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class StudentApiController : ControllerBase
    {
        private readonly SchoolDbContext _context;
        public StudentApiController(SchoolDbContext context) { _context = context; }

        // POST: api/StudentApi
        [HttpPost]
        public async Task<IActionResult> CreateStudent([FromBody] Student model)
        {
            // ១. ឆែកមើលក្រែង Client ផ្ញើ JSON ទទេមក
            if (model == null)
            {
                return BadRequest("ទិន្នន័យ JSON មិនអាចទទេបានឡើយ!"); // 400
            }

            // ២. សរសេរចូល SQL Server
            await _context.Students.AddAsync(model);
            await _context.SaveChangesAsync();

            // ៣. បញ្ចេញ HTTP 201 Created ភ្ជាប់ជាមួយ URL Resource ថ្មី
            return CreatedAtAction(nameof(GetStudentById), new { id = model.Id }, model);
        }

        // GET: api/StudentApi/5 (សម្រាប់ប្រើជាមួយ CreatedAtAction)
        [HttpGet("{id}")]
        public async Task<IActionResult> GetStudentById(int id)
        {
            var student = await _context.Students.FindAsync(id);
            if (student == null) return NotFound();
            return Ok(student);
        }
    }
}

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using SchoolApp.Data;
using SchoolApp.Models;

namespace SchoolApp.Controllers.Api 
{
    [Route("api/[controller]")]
    [ApiController]
    public class SupplierApiController : ControllerBase 
    {
        private readonly SchoolDbContext _context;

        public SupplierApiController(SchoolDbContext context) 
        { 
            _context = context; 
        }

        // POST: api/SupplierApi
        [HttpPost]
        public async Task<IActionResult> CreateSupplier([FromBody] Supplier model) 
        {
            if (!ModelState.IsValid) 
            {
                return BadRequest(ModelState); // 400 Bad Request
            }

            await _context.Suppliers.AddAsync(model);
            await _context.SaveChangesAsync();

            return CreatedAtAction(nameof(GetSupplierById), new { id = model.Id }, model); // 201 Created
        }

        // GET: api/SupplierApi/5
        [HttpGet("{id}")]
        public async Task<IActionResult> GetSupplierById(int id) 
        {
            var item = await _context.Suppliers.FindAsync(id);
            return item == null ? NotFound() : Ok(item);
        }
    }
}
