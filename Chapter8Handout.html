using Microsoft.AspNetCore.Mvc;
using SchoolApp.Models;

public class StudentController : Controller
{
    private readonly AppDbContext _context;
    public StudentController(AppDbContext context) { _context = context; }

    // ១. GET: បង្ហាញទម្រង់ Form ទទេ
    [HttpGet]
    public IActionResult Create()
    {
        return View(); // ហៅឯកសារ Create.cshtml មកបង្ហាញ
    }

    // ២. POST: ទទួលទិន្នន័យពី Form មក Save
    [HttpPost]
    [ValidateAntiForgeryToken] // ការពារការវាយប្រហារប្រភេទ CSRF
    public async Task<IActionResult> Create(Student model)
    {
        // ចាំបន្តនៅទំព័រក្រោយ (និយាយពី Validation)...
        
        await _context.Students.AddAsync(model);
        await _context.SaveChangesAsync(); // Save ចូល SQL
        
        // ក្រោយ Save រួច បញ្ជា Browser ឱ្យលោតទៅទំព័របញ្ជីឈ្មោះ (Index)
        return RedirectToAction(nameof(Index)); 
    }
}

[HttpPost]
[ValidateAntiForgeryToken]
public async Task<IActionResult> Create(Student model)
{
    // ឆែកកំហុសទិន្នន័យ (Server-Side Validation)
    if (!ModelState.IsValid)
    {
        // បើមានកំហុស (ឧ. អត់វាយឈ្មោះ) ហាម Save! 
        // ត្រូវបញ្ជូន Model ដដែល ត្រឡប់ទៅកាន់ View វិញ ដើម្បីបង្ហាញ Error ពណ៌ក្រហម
        return View(model); 
    }

    // បើ ModelState.IsValid ជាការពិត (True) នោះទើបយើង Save...
    await _context.Students.AddAsync(model);
    await _context.SaveChangesAsync();
    
    return RedirectToAction(nameof(Index));

[ApiController] // Attribute នេះជួយឆែក ModelState ដោយស្វ័យប្រវត្ត! (Auto 400 Bad Request)
public class StudentApiController : ControllerBase
{
    // ... (Constructor + DbContext) ...

    [HttpPost]
    // [FromBody] ចាប់យក JSON String ពី Client មកបំប្លែងជា C# Object 
    public async Task<IActionResult> CreateStudent([FromBody] Student model)
    {
        if (model == null)
            return BadRequest("ទិន្នន័យទទេ!");

        // Save ចូល Database ដូូច MVC ដែរ
        await _context.Students.AddAsync(model);

        return CreatedAtAction(nameof(GetStudentById), new { id = model.Id }, model);
    }
}

    // ១. បញ្ឈប់ការ Refresh!
    e.preventDefault();

        alert("ជោគជ័យ!");
        document.getElementById('apiStudentForm').reset(); // Clear Form
    } else { alert("ទិន្នន័យខុសទម្រង់ (400)"); }
});
</script>

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using SchoolApp.Data;
using SchoolApp.Models;

namespace SchoolApp.Controllers
{
    [Route("api/[controller]")]
    [ApiController] // បើក Auto-Validation (400 Bad Request)
    public class CategoryApiController : ControllerBase
    {
        private readonly AppDbContext _context;

        public CategoryApiController(AppDbContext context)
        {
            _context = context;
        }

        // ----------------------------------------------------
        // ១. [HttpPost] ទទួល JSON ពី Client មក Save
        // ----------------------------------------------------
        [HttpPost]
        public async Task<IActionResult> CreateCategory([FromBody] Category model)
        {
            // បញ្ជាក់៖ មិនបាច់សរសេរ if(!ModelState.IsValid) ទេ ព្រោះ API ធ្វើឱ្យរួចរាល់ហើយ

            if (model == null)
            {
                return BadRequest("No data received");
            }

            // Save ចូល DB
            await _context.Categories.AddAsync(model);
            await _context.SaveChangesAsync();
            
            // បញ្ចេញលទ្ធផល Status 201 ព្រមទាំង Object ថ្មីដែលមាន Id រួចរាល់
            return CreatedAtAction(nameof(GetCategoryById), new { id = model.Id }, model); 
        }

        // ----------------------------------------------------
        // ២. [HttpGet] Method សំរាប់ឱ្យ CreatedAtAction ហៅរក
        // ----------------------------------------------------
        [HttpGet("{id}")]
        public async Task<IActionResult> GetCategoryById(int id)
        {
            var category = await _context.Categories.FindAsync(id);
            if (category == null) return NotFound();
            return Ok(category);
        }
    }
}

                    body: JSON.stringify(newCat) // បំប្លែងទៅជា JSON
                });

                // ៤. ឆែកលទ្ធផល (201 Created)
                if (response.ok) {
                    alert("ជោគជ័យ! បញ្ចូលទិន្នន័យចូល Database រួចរាល់។");
                    catForm.reset(); // ជម្រះអត្ថបទក្នុងប្រអប់
                } else {
                    alert("បរាជ័យ (400 Bad Request)! សូមឆែកមើល Payload ក្នុងថេប Network។");
                }
            } catch(err) {
                console.error('បញ្ហាតភ្ជាប់ API:', err);
            }
        });
    </script>
</body>
</html>
