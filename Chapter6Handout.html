៦.២ ការប្រើប្រាស់ Data Annotations លើ Models
តាមលំនាំដើម (Convention) ប្រសិនបើអ្នកប្រកាស Property string Name; នោះ EF Core នឹងបង្កើត Column ក្នុង SQL ជាប្រភេទ nvarchar(max) ហើយអនុញ្ញាតឱ្យទទេ (Nullable)។ ប៉ុន្តែក្នុងការអនុវត្តជាក់ស្តែង យើងតែងតែចង់កំណត់លក្ខខណ្ឌដូចជា៖ ឈ្មោះមិនអាចទទេបាន (Not Null) និងមានប្រវែងត្រឹមតែ ៥០ តួអក្សរប៉ុណ្ណោះ។
ដើម្បីធ្វើដូចនេះបាន យើងត្រូវប្រើប្រាស់ Data Annotations ដែលជា Attributes សរសេរនៅពីលើ Property នីមួយៗ (ទាមទារ Library System.ComponentModel.DataAnnotations)។
តារាងសង្ខេប Data Annotations សំខាន់ៗ៖
Annotation Attribute តួនាទី និងការបកប្រែទៅកាន់ SQL Server
[Key] កំណត់ Property នោះឱ្យក្លាយជា Primary Key។ (ចំណាំ៖ បើ Property ឈ្មោះ Id វានឹងយកជា PK ស្វ័យប្រវត្តមិនបាច់ដាក់ក៏បាន)។
[Required] កំណត់មិនឱ្យទិន្នន័យទទេឡើយ គឺស្មើនឹងលក្ខខណ្ឌ NOT NULL នៅក្នុង SQL។
[MaxLength(50)] ឬ
[StringLength(50)]		 កំណត់ប្រវែងអតិបរមានៃអក្សរ។ វាបំប្លែងពី nvarchar(max) ទៅជា nvarchar(50) ជួយសន្សំសំចៃទំហំ Database។
[Column("col_name")] ប្រើសម្រាប់ប្តូរឈ្មោះ Column នៅក្នុង Database ឱ្យខុសពីឈ្មោះ Property ក្នុង C#។
[Table("tbl_Students")] (ដាក់លើក្បាល Class) ប្តូរឈ្មោះ Table ក្នុង SQL ជំនួសឱ្យការយកឈ្មោះតាម DbSet។

Models/Student.cs
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace SchoolApp.Models
{
    [Table("tbl_Students")] // បង្ខំឱ្យបង្កើត Table ឈ្មោះ tbl_Students
    public class Student
    {
        [Key]
        public int StudentId { get; set; } // កំណត់ជា Primary Key

        [Required(ErrorMessage = "ឈ្មោះមិនអាចទទេឡើយ")]
        [MaxLength(100)]
        [Column("FullName")]
        public string Name { get; set; } = string.Empty;

        [MaxLength(15)]
        public string? PhoneNumber { get; set; } // អនុញ្ញាតឱ្យ Null បាន (សញ្ញា ?)
    }
}

អត្ថប្រយោជន៍ទ្វេដង (Dual Benefits)

Data Annotations មិនត្រឹមតែត្រូវបាន EF Core យកទៅបំប្លែងជា Constraints (Not Null, varchar(100)) នៅក្នុង Database ប៉ុណ្ណោះទេ តែវាក៏ត្រូវបាន ASP.NET Core API យកទៅធ្វើជា Model Validation ដោយស្វ័យប្រវត្តផងដែរ។ មានន័យថា ប្រសិនបើ Client បញ្ជូនទិន្នន័យ (POST) មកដោយមិនមានបញ្ចូល Name នោះ API នឹងបដិសេធ (Return 400 Bad Request) ដោយស្វ័យប្រវត្ត មុននឹងវាទៅដល់ Database ម៉្លេះ។

using System.Collections.Generic;

public class Major
{
    public int Id { get; set; }
    public string Name { get; set; } = string.Empty;

    // Navigation Property: ជំនាញមួយ មាននិស្សិតច្រើននាក់
    public List<Student> Students { get; set; } = new List<Student>();
}
Models/Student.cs (The "Many" side)
public class Student
{
    public int Id { get; set; }
    public string Name { get; set; } = string.Empty;

    // Foreign Key: ត្រូវមានឈ្មោះជា {NameOfReferenceClass}Id
    public int MajorId { get; set; }

    // Navigation Property: និស្សិតម្នាក់ ეკუთვნឹងជំនាញមួយ
    public Major? Major { get; set; } 
}

តួនាទី៖ បញ្ជានេះនឹងស្កេនមើលរាល់ការផ្លាស់ប្តូរនៅក្នុង Models របស់អ្នក (ធៀបនឹងពេលមុន) រួចបង្កើតឯកសារ C# ថ្មីមួយនៅក្នុង Folder Migrations។ ឯកសារនេះផ្ទុកនូវកូដសម្រាប់ត្រៀមបង្កើត Tables (វាប្រៀបដូចជាសេចក្តីព្រាងប្លង់មេ មិនទាន់ប៉ះពាល់ Database ឡើយ)។
ឧទាហរណ៍៖ dotnet ef migrations add InitialCreate

2. dotnet ef database update

តួនាទី៖ បញ្ជានេះនឹងយកឯកសារ Migration ដែលទើបបង្កើត រួចបកប្រែវាទៅជាភាសា SQL Scripts (CREATE TABLE, ALTER TABLE...) ហើយបញ្ជូនវាទៅប្រតិបត្តិនៅក្នុង SQL Server ផ្ទាល់តែម្តង។ ពេល Command នេះរត់ចប់ Database របស់អ្នកនឹងត្រូវបានបង្កើត ឬ Update ជោគជ័យ។

សេចក្តីសង្ខេបជំពូកទី ៦ (Summary)
  • កម្លាំងនៃ Code-First៖ Code-First ជួយឱ្យអ្នកអភិវឌ្ឍន៍អាចគ្រប់គ្រងទាំង Business Logic និងរចនាសម្ព័ន្ធ Database (Schema) ពីកន្លែងតែមួយគឺកូដ C#។ វាសាកសមបំផុតសម្រាប់ Agile Development ។
  • សុវត្ថិភាពទិន្នន័យ (Data Integrity)៖ ការប្រើប្រាស់ Data Annotations ដូចជា [Required] និង [MaxLength] ការពារមិនឱ្យទិន្នន័យខុសទម្រង់ចូលទៅក្នុង Database ព្រមទាំងជួយសន្សំសំចៃទំហំ Disk Space (ឧ. កំណត់ nvarchar ឱ្យមានទំហំជាក់លាក់)។
  • ភាពងាយស្រួលនៃ Relationships៖ EF Core មានភាពឆ្លាតវៃក្នុងការរៀបចំ Foreign Keys ដោយស្វ័យប្រវត្ត នៅពេលយើងប្រកាស Navigation Properties ត្រឹមត្រូវ។ វាងាយស្រួលជាងការសរសេរ JOIN Constraints ក្នុង SQL ឆ្ងាយណាស់។
  • យន្តការ Migrations៖ ជាស្ពានភ្ជាប់តែមួយគត់រវាង C# Models និង SQL Server ។ ចងចាំជានិច្ចនូវលំដាប់លំដោយ៖ កែប្រែ Model → រត់បញ្ជា Add Migration → រត់បញ្ជា Update Database។

១. សំណួរ ខុស/ត្រូវ (True/False)

សេចក្តីណែនាំ៖ សូមគូសសញ្ញា (ត្រូវ) ប្រសិនបើប្រយោគខាងក្រោមត្រឹមត្រូវ ឬសញ្ញា (ខុស) ប្រសិនបើខុស។

  1. វិធីសាស្ត្រ Code-First តម្រូវឱ្យអ្នកមាន Database និង Tables នៅក្នុង SQL Server ជាមុន ទើបអាចសរសេរកូដ C# បាន។
  2. Annotation [Required] ប្រើសម្រាប់កំណត់ Column នៅក្នុង Database មិនឱ្យមានតម្លៃ Null (NOT NULL)។
  3. ប្រសិនបើ Property មានឈ្មោះថា Id នោះ EF Core នឹងកំណត់វាជា Primary Key ដោយស្វ័យប្រវត្តដោយមិនបាច់ប្រើ [Key] ឡើយ។
  4. ពាក្យបញ្ជា dotnet ef migrations add នឹងធ្វើការបង្កើត ឬកែប្រែ Tables នៅក្នុង SQL Server ភ្លាមៗតែម្តង។
  5. ការប្រកាស List<Student> Students នៅក្នុង Class Major គឺជាឧទាហរណ៍នៃ Navigation Property ។
  6. តារាង __EFMigrationsHistory ប្រើសម្រាប់ផ្ទុកទិន្នន័យ (Data) របស់ User ដូចជាឈ្មោះ និងលេខទូរស័ព្ទ។
  7. Annotation [Table("tbl_Users")] ជួយប្តូរឈ្មោះ Table នៅក្នុង Database ឱ្យខុសពីឈ្មោះ Class ឬឈ្មោះ DbSet ។
  8. ដើម្បីបង្កើតទំនាក់ទំនង One-to-Many ខាងតារាង "Many" ត្រូវតែមាន Foreign Key Property ។
  9. អ្នកត្រូវតែរត់ពាក្យបញ្ជា dotnet ef database update រាល់ពេលដែលអ្នកបន្ថែម Migration ថ្មី ដើម្បីឱ្យ SQL Server ស្គាល់ការកែប្រែ។

  • ជំហានទី ១ (បង្កើត Major Model)៖ ចូរបង្កើតឯកសារ Models/Major.cs ។ ប្រើ [Table("tbl_Majors")]។ ប្រកាស Property: Id (ជា [Key]), Title (ជា [Required] និង [MaxLength(50)]), ព្រមទាំង Navigation Property List<Student> ។

using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace SchoolApp.Models

    [Table("tbl_Majors")]
    public class Major
    {
        [Key]
        public int Id { get; set; }

        [Required]
        [MaxLength(50)]

        // Navigation Property (មួយ Major មាននិស្សិតច្រើន)
        public List<Student> Students { get; set; } = new List<Student>();
    }
}

using Microsoft.EntityFrameworkCore;
using SchoolApp.Models;

namespace SchoolApp.Data
{
    public class SchoolDbContext : DbContext
    {
        public SchoolDbContext(DbContextOptions<SchoolDbContext> options) : base(options) {}

        // ប្រកាស Tables ទាំងពីរ
        public DbSet<Major> Majors { get; set; }
        public DbSet<Student> Students { get; set; }
    }
}
៤️⃣ ជំហានទី ៤៖ កំណត់ Connection និងចុះឈ្មោះ DI Container
appsettings.json (បន្ថែមផ្នែក ConnectionStrings)
"ConnectionStrings": {
    "SchoolConnection": "Server=localhost\\SQLEXPRESS;Database=SchoolDB_Migrations;Trusted_Connection=True;TrustServerCertificate=True;"
}
Program.cs
using Microsoft.EntityFrameworkCore;
using SchoolApp.Data;

var builder = WebApplication.CreateBuilder(args);

// ទាញយក ConnectionString ពីឯកសារកំណត់រចនាសម្ព័ន្ធ
var connectionString = builder.Configuration.GetConnectionString("SchoolConnection");

// ចុះឈ្មោះ SchoolDbContext ជាមួយ SQL Server
builder.Services.AddDbContext<SchoolDbContext>(options =>
    options.UseSqlServer(connectionString));

// (កូដ Controller បន្តបន្ទាប់...)

ចំណេះដឹងបន្ថែម (Bonus Workflow)៖ ការទាញទិន្នន័យដែលមានទំនាក់ទំនង (Include)
នៅពេលយើងមានទំនាក់ទំនង One-to-Many បើយើងទាញយកត្រឹមតែ _context.Students.ToList() នោះវានឹងចេញតែឈ្មោះនិស្សិត និង MajorId (លេខ) ប៉ុណ្ណោះ វាលាក់ព័ត៌មានលម្អិតរបស់ Major ចោល។ ដើម្បីភ្ជាប់ទិន្នន័យ (ស្មើនឹង SQL JOIN) នៅក្នុង EF Core យើងត្រូវប្រើ Method .Include()។
ឧទាហរណ៍នៃការសរសេរ Controller ដើម្បីហៅទិន្នន័យដែលមាន JOIN៖
Controllers/StudentApiController.cs
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore; // ត្រូវការសម្រាប់ប្រើ .Include()
using SchoolApp.Data;

namespace SchoolApp.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class StudentApiController : ControllerBase
    {
        private readonly SchoolDbContext _context;

        public StudentApiController(SchoolDbContext context) { _context = context; }

        [HttpGet]
        public IActionResult GetAllStudentsWithMajors()
        {
            // .Include(s => s.Major) ដើរតួដូច SQL INNER JOIN ទៅកាន់ Table Majors
            var students = _context.Students
                                   .Include(s => s.Major) 
                                   .ToList();

            return Ok(students);
        }
    }
}
លទ្ធផល JSON ដែលរំពឹងទុក៖ វានឹងបង្ហាញឈ្មោះសិស្ស និងមានភ្ជាប់នូវ Object នៃ Major ទាំងមូល (មានទាំង Id និង Title របស់ជំនាញ) ដែលផ្តល់ភាពងាយស្រួលបំផុតដល់ Frontend Developer យកទៅប្រើប្រាស់។ (មេរៀនលម្អិតពី GET API នឹងមានក្នុងជំពូកបន្ទាប់)។
