using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using SchoolApp.Data;

namespace SchoolApp.Controllers
{
    // ស្នងពី Controller ធម្មតា (មាន View Support)
    public class StudentController : Controller
    {
        private readonly SchoolDbContext _context;

        public StudentController(SchoolDbContext context)
        {
            _context = context;
        }

        // GET: /Student/Index
        public async Task<IActionResult> Index()
        {
            // ១. ទាញយកទិន្នន័យសិស្សពី SQL Server

            return View(students);
        }
    }
}

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using SchoolApp.Data;

namespace SchoolApp.Controllers
{
    [Route("api/[controller]")] // Endpoint: api/StudentApi
    [ApiController]               // បញ្ជាក់ថាជា Web API Controller
    public class StudentApiController : ControllerBase // មិនត្រូវការ View Support ឡើយ
    {
        private readonly SchoolDbContext _context;

        public StudentApiController(SchoolDbContext context)
        {
            _context = context;
        }

        [HttpGet] // HTTP GET Request
        public async Task<IActionResult> GetStudents()
        {
            var students = await _context.Students.ToListAsync();

            if (students == null || students.Count == 0)
            {
                return NotFound("រកមិនឃើញទិន្នន័យសិស្សឡើយ!"); // Status Code 404
            }

            return Ok(students); // Status Code 200 + JSON Data
        }
    }
}

• Traditional Controller ត្រឡប់៖ View(students) → ផលិតចេញជា HTML Page ធំ។

ការកំណត់ CORS នៅក្នុង Program.cs (Backend) 3. Apply
Program.cs
var builder = WebApplication.CreateBuilder(args);

// ១. ចុះឈ្មោះសេវាកម្ម CORS ឱ្យស្គាល់គ្រប់ប្រភព (សម្រាប់ Development)
builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowAll", policy =>
    {
        policy.AllowAnyOrigin()  // អនុញ្ញាតគ្រប់ Domain / Port
              .AllowAnyHeader()  // អនុញ្ញាតរាល់ Headers
              .AllowAnyMethod(); // អនុញ្ញាត GET, POST, PUT, DELETE
    });
});

var app = builder.Build();

// ២. ដាក់ឱ្យដំណើរការ CORS (ត្រូវតែនៅពីលើ UseAuthorization)
app.UseCors("AllowAll");
app.UseAuthorization();
app.MapControllers();
app.Run();

ក្នុងការអភិវឌ្ឍន៍ជាក់ស្តែង កំហុសដែលកើតឡើងលើ Traditional MVC និង Decoupled Web API មានលក្ខណៈខុសគ្នាទាំងស្រុង៖

1. Bugs លើ Traditional MVC

  • NullReferenceException: កើតឡើងពេលបោះ Model ទៅ View តែ Model គ្មានទិន្នន័យ (Null)។
    ដំណោះស្រាយ៖ ឆែកមើល @if(Model != null) ក្នុង Razor View។
  • InvalidOperationException: Model Type ក្នុង `.cshtml` (ឧ. `Student`) មិនត្រូវគ្នានឹង Controller (`List`)។

2. Bugs លើ Web API & Fetch

  • CORS Error: លោតអក្សរក្រហមក្នុង Console "No Access-Control-Allow-Origin"。
    ដំណោះស្រាយ៖ ឆែកកូដ app.UseCors() ក្នុង Program.cs។
  • Undefined Fields: តារាងចេញពាក្យ undefined មកពីសរសេរ s.FullName (PascalCase) ជំនួសឱ្យ s.fullName (camelCase)។

// ប្រើ console.log ដើម្បីពិនិត្យមើលទម្រង់ JSON ជាក់ស្តែងដែលបានមកពី API
const response = await fetch(apiUrl);
const data = await response.json();
console.log("Data from API:", data); // បើកមើលក្នុង F12 Console របស់ Browser

async function loadProductsAdvanced() {
    try {
        const res = await fetch('https://localhost:7123/api/ProductApi');

        // ឆែកមើល Status Code ប្រសិនបើមិនមែន 200-299
        if (!res.ok) {
            if (res.status === 404) throw new Error("រកមិនឃើញ Endpoint ឬទិន្នន័យឡើយ (404)");
            if (res.status === 500) throw new Error("កំហុសបច្ចេកទេសលើ Server (500)");
            throw new Error(`HTTP Error: ${res.status}`);
        }

        const data = await res.json();
        console.log("ទិន្នន័យទទួលបានជោគជ័យ:", data);

    } catch (err) {
        alert(err.message);
    }
}
