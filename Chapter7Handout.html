        // ទទួលសំណើប្រភេទ GET (គ្មាន Parameter)
        [HttpGet]
        // ប្រើ async Task និងរុំលទ្ធផលក្នុង <IActionResult>
        public async Task<IActionResult> GetStudents()
        {
            // ១. ប្រើ await ដើម្បីរង់ចាំលទ្ធផលពី Database យ៉ាងមានប្រសិទ្ធភាព
            // (តម្រូវឱ្យ using Microsoft.EntityFrameworkCore;)
            var studentsList = await _context.Students.ToListAsync();

            // ២. ឆែកមើលថាតើមានទិន្នន័យឬទេ
            if (studentsList == null || studentsList.Count == 0)
            {
                return NotFound("មិនមានទិន្នន័យសិស្សនៅក្នុងប្រព័ន្ធឡើយ។"); // Status 404
            }

            // ៣. បញ្ចេញទិន្នន័យជា JSON ភ្ជាប់ជាមួយ Status 200 OK
            return Ok(studentsList);
        }

៧.៤ ការសាកល្បង API ដោយប្រើប្រាស់ Swagger UI

  2. បើក Browser រួចវាយអាសយដ្ឋាន ដែលបញ្ចប់ដោយពាក្យ swagger។
    ឧទាហរណ៍៖ https://localhost:7123/swagger

៧.៥ ការដោះស្រាយបញ្ហាសុវត្ថិភាព CORS Policy
មុននឹងឈានទៅសរសេរកូដ Frontend យើងត្រូវដោះស្រាយរបាំងសុវត្ថិភាពមួយសិន។ Web Browser មានច្បាប់មួយហៅថា Same-Origin Policy (SOP) ដែលហាមឃាត់មិនឱ្យ Website មួយ (ឧ. រត់លើ Live Server http://127.0.0.1:5500) លួចហៅទៅយកទិន្នន័យពី Server មួយផ្សេងទៀត (ឧ. https://localhost:7123) ឡើយ។
ប្រសិនបើយើងមិនដោះស្រាយទេ ពេលសរសេរកូដ JS ទាញទិន្នន័យ វានឹងលោត Error ពណ៌ក្រហមពាក់ព័ន្ធនឹងពាក្យ "CORS policy: No 'Access-Control-Allow-Origin' header is present" នៅក្នុង Console របស់ Browser ជាមិនខាន។ ដំណោះស្រាយគឺយើងត្រូវកំណត់ CORS (Cross-Origin Resource Sharing) នៅក្នុងគម្រោង Backend។
ការកំណត់ CORS នៅក្នុងឯកសារ Program.cs (Backend)
Program.cs
var builder = WebApplication.CreateBuilder(args);

// ១. ចុះឈ្មោះសេវាកម្ម CORS ឱ្យស្គាល់គ្រប់ប្រភព (សម្រាប់ Development)
builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowAll", policy =>
    {
        policy.AllowAnyOrigin()  // អនុញ្ញាតពីគ្រប់ Domain (Port)
              .AllowAnyHeader()  // អនុញ្ញាតរាល់ Headers ទាំងអស់
              .AllowAnyMethod(); // អនុញ្ញាត GET, POST, PUT, DELETE
    });
});

// ... កូដផ្សេងៗដូចជា AddDbContext និង AddControllers ...
var app = builder.Build();

// ២. ដាក់ឱ្យដំណើរការ CORS Policy (ចំណាំ៖ ត្រូវតែសរសេរនៅខាងលើ UseAuthorization)
app.UseCors("AllowAll");

app.UseAuthorization();
app.MapControllers();
app.Run();
* ក្រោយពីកែប្រែ Program.cs រួច សូមកុំភ្លេច Restart Server (បិទ Terminal រួច dotnet run ឡើងវិញ) ដើម្បីឱ្យការកំណត់ថ្មីមានប្រសិទ្ធភាព។

៧.៧ ការប្រើប្រាស់ JS Fetch (Async / Await)
បច្ចេកវិទ្យាដែលយើងប្រើសម្រាប់ទាក់ទងជាមួយ Backend គឺមុខងារ fetch() របស់ JavaScript។ នៅក្នុងកម្រិតវិជ្ជាជីវៈ យើងប្រើប្រាស់យន្តការ async / await ជំនួសឱ្យ .then() ដើម្បីធ្វើឱ្យកូដមើលទៅស្អាត ងាយយល់ និងងាយស្រួលគ្រប់គ្រង Error តាមរយៈ `try...catch`។
សូមសរសេរកូដ <script> នេះនៅខាងក្រោមតារាងក្នុងឯកសារ index.html ដដែល (មុនបិទ </body>)។
index.html (ផ្នែក JavaScript)
<script>
    // កំណត់ URL របស់ Backend API របស់អ្នក (ដូរ Port ឱ្យត្រូវនឹងគម្រោងអ្នក)
    const apiUrl = 'https://localhost:7123/api/StudentApi';

    // ចាប់យកប៊ូតុង និងកន្លែងត្រូវបញ្ចេញទិន្នន័យ (tbody)
    const btnLoad = document.getElementById('btnLoad');
    const tableBody = document.getElementById('tableBody');

    // ភ្ជាប់ការចុចប៊ូតុងទៅនឹងអនុគមន៍អសមកាល (async function)
    btnLoad.addEventListener('click', async function() {
        try {
            // ១. ហៅទៅកាន់ API ហើយ "await" រង់ចាំលទ្ធផលត្រឡប់មកវិញសិន
            const response = await fetch(apiUrl);
            
            // ២. បំប្លែងទិន្នន័យអត្ថបទទៅជា JS Array Object (រង់ចាំការបំប្លែង)
            const data = await response.json();

            // ៣. លុបទិន្នន័យចាស់ចោលសិន មុននឹងបញ្ចាំងថ្មី (ជៀសវាងលោតជាន់គ្នា)
            tableBody.innerHTML = '';
            
            // ៤. ដំណើរការ Loop លើទិន្នន័យដែលបានមក ដើម្បីបង្កើត HTML Row ថ្មី
            data.forEach(student => {
                const row = `
                    <tr>
                        <td>${student.id}</td>
                        <td>${student.fullName}</td>
                    </tr>
                `;
                // បញ្ចូលជួរឈរ (row) ទៅក្នុងតារាង html ផ្ទាល់ (DOM Manipulation)
                tableBody.innerHTML += row;
            });

        } catch (error) {
            // ៥. ចាប់កំហុស (ឧ. Server បិទ ឬជាប់បញ្ហា CORS)
            console.error('Error fetching data:', error);
            alert("មិនអាចទាញទិន្នន័យបានទេ សូមពិនិត្យមើល Console!");
        }
    });
</script>

របៀបប្រើប្រាស់ Inspect Element (F12) លើ Chrome៖

  1. នៅលើទំព័រ Web របស់អ្នក សូមចុចគ្រាប់ចុច F12 ឬចុចស្តាំយកពាក្យ Inspect។
  2. ចូលទៅកាន់ថេប Console៖
    • ប្រសិនបើមានបញ្ហាដូចជា Error កូដ JavaScript សរសេរខុស ឬបញ្ហា CORS វានឹងបង្ហាញអក្សរពណ៌ក្រហមនៅទីនេះ។
    • អ្នកក៏អាចប្រើ console.log(data) ក្នុងកូដ JS របស់អ្នក ដើម្បីបញ្ចាំងមើលទិន្នន័យ JSON នៅក្នុងថេបនេះផងដែរ។
  3. ចូលទៅកាន់ថេប Network៖
    • ថេបនេះសម្រាប់តាមដានរាល់សំណើ HTTP (API Requests) ដែលបានបាញ់ចេញទៅ Server។
    • ពេលអ្នកចុចប៊ូតុង "ទាញយកទិន្នន័យ" អ្នកនឹងឃើញឈ្មោះ API របស់អ្នកលេចឡើងទីនេះ។ ចុចលើវា រួចរើសថេប Response ដើម្បីមើលថាតើ Backend ពិតជាបានឆ្លើយតបលទ្ធផល JSON ត្រឡប់មកវិញឬទេ។

១. សំណួរ ខុស/ត្រូវ (True/False)

សេចក្តីណែនាំ៖ សូមគូសសញ្ញា (ត្រូវ) ប្រសិនបើប្រយោគខាងក្រោមត្រឹមត្រូវ ឬសញ្ញា (ខុស) ប្រសិនបើខុស។

  1. Traditional MVC គឺជាទម្រង់ដែល Backend គ្រាន់តែបញ្ចេញទិន្នន័យជា JSON ប៉ុណ្ណោះ ចំណែក Frontend ជាអ្នករៀបចំ HTML ខ្លួនឯង។
  2. HTTP GET Method ត្រូវបានប្រើប្រាស់ជាចម្បងដើម្បីអាន ឬទាញយកទិន្នន័យពីប្រព័ន្ធ Database តាមរយៈ API។

  4. មុខងារ fetch() នៅក្នុង JavaScript នឹងតម្រូវឱ្យទំព័រ Web Refresh (Reload) ខ្លួនឯងរាល់ពេលដែលវាទាញទិន្នន័យ។
  5. ការប្រើ return Ok(studentsList); ក្នុង C# នឹងបញ្ចេញលទ្ធផលអមជាមួយ HTTP Status Code 200 (ជោគជ័យ) ព្រមទាំងទិន្នន័យ JSON ។
  6. CORS Error ច្រើនតែកើតឡើងនៅពេលដែល Frontend (Web Page) និង Backend API ស្ថិតនៅលើ Port ឬ Domain ខុសគ្នា។
  7. Swagger គឺជាកម្មវិធីកម្ចាត់មេរោគដែលភ្ជាប់មកជាមួយ .NET Core ដើម្បីការពារកុំឱ្យ Hacker លួចចូល Database។
  8. នៅក្នុង JavaScript (DOM) យើងអាចប្រើប្រាស់ tableBody.innerHTML += row ដើម្បីបន្ថែមជួរឈរថ្មីចូលក្នុងតារាង។
  9. ទម្រង់ Naming Convention របស់ JSON ដែលបញ្ចេញពី .NET ជាធម្មតានឹងប្តូរអក្សរធំនៅដើម (PascalCase) ទៅជាអក្សរតូច (camelCase) ដោយស្វ័យប្រវត្ត។
  10. កូដ await response.json() មានតួនាទីបំប្លែងទិន្នន័យអត្ថបទ JSON ពី Server ឱ្យក្លាយជា Array/Object របស់ JavaScript។

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using SchoolApp.Data;

namespace SchoolApp.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ProductApiController : ControllerBase
    {
        private readonly AppDbContext _context;

        public ProductApiController(AppDbContext context) { _context = context; }

        [HttpGet]
        // ការប្រើប្រាស់ក្បួន Async ដើម្បីប្រសិទ្ធភាពខ្ពស់
        public async Task<IActionResult> GetProducts()
        {
            // ទាញទិន្នន័យទំនិញទាំងអស់ចេញពី Database (ដោយមិនគាំង Server)
            var products = await _context.Products.ToListAsync();
            
            // បញ្ចេញជា JSON Status 200 OK
            return Ok(products);
        }
    }
}

ការហៅទិន្នន័យតាមរយៈអ៊ីនធឺណិតមិនលឿនដូចការអានពី Hard Disk កុំព្យូទ័រផ្ទាល់ទេ។ ជាញឹកញាប់ វាអាចចំណាយពេលពី ១ ទៅ ២ វិនាទី (Latency)។ ដើម្បីធ្វើឱ្យកម្មវិធីរបស់អ្នកកាន់តែមានលក្ខណៈអាជីព (Professional UX) យើងគួរតែបង្ហាញពាក្យថា "កំពុងទាញយកទិន្នន័យ (Loading...)" ពេលដែលកំពុង Fetch និងលុបវាចេញវិញពេលការងារត្រូវបានបញ្ចប់។
ឧទាហរណ៍នៃការប្រើប្រាស់ block try/catch/finally នៅក្នុង JS៖
products.html (កែប្រែ JS បន្តិច)
<script>
    btnFetchData.addEventListener('click', async function() {
        
        // បង្ហាញ Loading មុនពេលចាប់ផ្តើមហៅ API
        productBox.innerHTML = '<tr><td colspan="3" style="text-align:center">កំពុងដំណើរការ (Loading)...</td></tr>';
        btnFetchData.disabled = true; // បិទប៊ូតុងកុំឱ្យ User ចុចផ្ទួនៗ

        try {
            const res = await fetch('https://localhost:7123/api/ProductApi');
            const data = await res.json();
                
            productBox.innerHTML = ''; // លុបអក្សរ Loading ចេញ ពេលទិន្នន័យមកដល់

            data.forEach(item => {
                productBox.innerHTML += `<tr><td>${item.id}</td><td>${item.name}</td><td>${item.price}</td></tr>`;
            });

        } catch (err) {
            // បើមានបញ្ហា(Error) បង្ហាញពាក្យនេះជំនួសវិញ
            productBox.innerHTML = '<tr><td colspan="3" style="color:red; text-align:center">បរាជ័យក្នុងការតភ្ជាប់ API ផ្ទៀងផ្ទាត់ CORS ឬ Server!</td></tr>';
            console.error(err);
        } finally {
            // Block នេះនឹងរត់ជានិច្ច មិនថាជោគជ័យឬបរាជ័យ
            btnFetchData.disabled = false; // បើកប៊ូតុងឱ្យប្រើវិញ
        }
    });
</script>
