using (var conn = new SqlConnection(connStr)) {
    conn.Open();
    var cmd = new SqlCommand("SELECT * FROM Students", conn);
    using (var r = cmd.ExecuteReader()) {
        while (r.Read()) {
            var s = new Student {
                Id = (int)r["Id"],
                Name = r["Name"].ToString()
            };
            students.Add(s);
        }
    }
}

៥.៣ រចនាសម្ព័ន្ធ Connection String ក្នុង appsettings.json វិភាគ - Analyze
Connection String គឺជាខ្សែអត្ថបទដែលផ្ទុកនូវព័ត៌មានលម្អិតដូចជា ទីតាំង Server, ឈ្មោះ Database និងសិទ្ធិអនុញ្ញាត (Credentials) ដើម្បីឱ្យ API អាចតភ្ជាប់ទៅកាន់ Database បាន។ តាមស្តង់ដារសុវត្ថិភាព យើងត្រូវសរសេរវានៅក្នុងឯកសារ appsettings.json។
រចនាសម្ព័ន្ធកូដក្នុង appsettings.json៖
appsettings.json
{
  "Logging": { ... },
  "AllowedHosts": "*",
  "ConnectionStrings": {
    "DefaultConnection": "Server=localhost\\SQLEXPRESS;Database=SchoolDB;Trusted_Connection=True;TrustServerCertificate=True;"
  }
}
ការពន្យល់ពីប៉ារ៉ាមែត្រនីមួយៗនៃ Connection String៖
ប៉ារ៉ាមែត្រ (Parameter) អត្ថន័យ និងការប្រើប្រាស់ជាក់ស្តែង
Server=
localhost\\SQLEXPRESS;		 បញ្ជាក់ពីទីតាំងម៉ាស៊ីន និងឈ្មោះ Instance របស់ SQL Server។ សញ្ញា \\ ត្រូវបានប្រើដើម្បី Escape Character នៅក្នុងទម្រង់ JSON។
Database=
SchoolDB;		 បញ្ជាក់ពីឈ្មោះមូលដ្ឋានទិន្នន័យ (Database Name) ដែលយើងចង់បង្កើត ឬចង់ប្រើប្រាស់។
Trusted_Connection=
True;		 ប្រើប្រាស់គណនី Windows (Windows Authentication) ដើម្បីតភ្ជាប់ដោយមិនបាច់វាយបញ្ចូល Username និង Password របស់ SQL ឡើយ។
TrustServerCertificate=
True;		 អនុញ្ញាតឱ្យរំលងការត្រួតពិនិត្យ SSL Certificate របស់ SQL Server សម្រាប់ការអភិវឌ្ឍន៍នៅម៉ាស៊ីន Local។

៥.៤ ការកសាង Class DbContext និងការប្រើប្រាស់ DbSet អនុវត្ត - Apply
DbContext គឺជា Class ស្នូលរបស់ EF Core ដែលតំណាងឱ្យការបើកសម័យកាលធ្វើការ (Session) មួយរវាងកម្មវិធី C# និង Database SQL Server។ រាល់តារាង (Tables) ដែលយើងចង់បង្កើតនៅក្នុង DB ត្រូវតែប្រកាសជា Property ប្រភេទ DbSet<TEntity> នៅក្នុង Class នេះ។
រចនាសម្ព័ន្ធកូដ AppDbContext ពេញលេញ៖
Data/AppDbContext.cs
using Microsoft.EntityFrameworkCore;
using SchoolApp.Models;

namespace SchoolApp.Data
{
    // ត្រូវតែស្នងត្រកូលពី class DbContext របស់ EF Core
    public class AppDbContext : DbContext
    {
        // Constructor សម្រាប់ទទួលយកការកំណត់ (Connection String) ពី Program.cs
        public AppDbContext(DbContextOptions<AppDbContext> options) : base(options)
        {
        }

        // ប្រកាស DbSet: Class 'Student' នឹងត្រូវបម្លែងជា Table ឈ្មោះ 'Students'
        public DbSet<Student> Students { get; set; }
    }
}
យន្តការបម្លែងទិន្នន័យស្វ័យប្រវត្ត (Convention Over Configuration)៖
  • Table Names៖ ឈ្មោះ Property របស់ DbSet (ឧ. Students) នឹងត្រូវបានយកទៅដាក់ជាឈ្មោះ Table នៅក្នុង SQL Server ដោយស្វ័យប្រវត្ត។
  • Primary Keys៖ Property នៅក្នុង Model ដែលមានឈ្មោះ Id ឬ StudentId នឹងត្រូវបានសម្គាល់ជា Primary Key (Identity) ដែលកើនឡើងលេខរៀងដោយខ្លួនឯងដោយស្វ័យប្រវត្ត។
  • Data Types៖ ប្រភេទ string ក្នុង C# នឹងត្រូវបានបម្លែងទៅជា nvarchar(max) នៅក្នុង SQL។

៥.៥ ការចុះឈ្មោះ DbContext តាមរយៈ Dependency Injection វាយតម្លៃ - Evaluate
Dependency Injection (DI) គឺជាបច្ចេកទេសរចនាកូដ (Design Pattern) ដែលប្រើដើម្បីកាត់បន្ថយភាពចងភ្ជាប់គ្នា (Loose Coupling)។ នៅក្នុង ASP.NET Core យើងមិនប្រើពាក្យ new AppDbContext() គ្រប់ទីកន្លែងឡើយ តែយើងចុះឈ្មោះវាចូលទៅក្នុង DI Container នៃ Program.cs ដើម្បីឱ្យប្រព័ន្ធជាអ្នកបង្កើត និងបញ្ជូនវាទៅកាន់ Controller ដោយស្វ័យប្រវត្ត។
ការចុះឈ្មោះ AppDbContext នៅក្នុងឯកសារ Program.cs ៖
Program.cs
using Microsoft.EntityFrameworkCore;
using SchoolApp.Data;

var builder = WebApplication.CreateBuilder(args);

// ១. ចាប់យកខ្សែស្រឡាយតភ្ជាប់ពីឯកសារ appsettings.json
var connectionString = builder.Configuration.GetConnectionString("DefaultConnection");

// ២. ចុះឈ្មោះ AppDbContext ទៅក្នុង DI Container ដោយប្រាប់វាឱ្យប្រើ SQL Server
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseSqlServer(connectionString));

builder.Services.AddControllers();
var app = builder.Build();

app.UseAuthorization();
app.MapControllers();
app.Run();

អាយុកាលសេវាកម្ម (Service Lifetime) របស់ DbContext

នៅពេលយើងប្រើប្រាស់ Method AddDbContext() ប្រព័ន្ធ .NET នឹងចុះឈ្មោះ AppDbContext ជាប្រភេទអាយុកាល Scoped Lifetime ដោយស្វ័យប្រវត្ត។ មានន័យថា ប្រព័ន្ធនឹងបង្កើត Object DbContext ថ្មីមួយ សម្រាប់រាល់ HTTP Request នីមួយៗ (ធានាសុវត្ថិភាពទិន្នន័យមិនឱ្យឆ្លងគ្នា)។

using Microsoft.EntityFrameworkCore;
using MyApp.Models;

namespace MyApp.Data
{
    public class StoreDbContext : DbContext
    {
        public StoreDbContext(DbContextOptions<StoreDbContext> options) : base(options)
        {
        }

        public DbSet<Category> Categories { get; set; }
    }
}

ដំណោះស្រាយ៖ ផ្នែកទី ១ (បន្ត)
៣️⃣ ជំហានទី ៣៖ កំណត់ Connection String, ចុះឈ្មោះ DI និងបើកសិទ្ធិ CORS
appsettings.json
"ConnectionStrings": {
    "StoreConnection": "Server=localhost\\SQLEXPRESS;Database=MyStoreDB;Trusted_Connection=True;TrustServerCertificate=True;"
}
Program.cs (Configuration & CORS Setup)
using Microsoft.EntityFrameworkCore;
using MyApp.Data;

var builder = WebApplication.CreateBuilder(args);

// [ចំណុចទី ១]. ចាប់យក Connection និងចុះឈ្មោះចូល DI Container
var conn = builder.Configuration.GetConnectionString("StoreConnection");
builder.Services.AddDbContext<StoreDbContext>(opt => opt.UseSqlServer(conn));

// [ចំណុចទី ២]. ដោះស្រាយបញ្ហា CORS (អនុញ្ញាតឱ្យ Frontend ហៅ API ចូលបាន)
builder.Services.AddCors(options => {
    options.AddPolicy("AllowAll", policy => {
        policy.AllowAnyOrigin().AllowAnyMethod().AllowAnyHeader();
    });
});

builder.Services.AddControllers();
var app = builder.Build();

// [ចំណុចទី ៣]. បើកដំណើរការ CORS Policy ដែលទើបកំណត់ខាងលើ
app.UseCors("AllowAll");

app.MapControllers();
app.Run();

ដំណោះស្រាយ៖ ផ្នែកទី ១ (បញ្ចប់ API)
៤️⃣ ជំហានទី ៤៖ បង្កើត Controller បញ្ចេញទិន្នន័យ JSON
កូដនេះបង្ហាញពីការប្រើប្រាស់ Dependency Injection ដើម្បីហៅ StoreDbContext មកប្រើក្នុង Controller រួចទាញទិន្នន័យពី Database បញ្ចេញជា JSON ទៅកាន់ View វិញ។
Controllers/CategoryApiController.cs
using Microsoft.AspNetCore.Mvc;
using MyApp.Data;

namespace MyApp.Controllers
{
    [Route("api/[controller]")] // URL ដែលត្រូវហៅគឺ: /api/categoryapi
    [ApiController]
    public class CategoryApiController : ControllerBase
    {
        private readonly StoreDbContext _context;

        // Constructor ទទួលយក DbContext តាមរយៈ DI
        public CategoryApiController(StoreDbContext context)
        {
            _context = context;
        }

        // Endpoint សម្រាប់ទាញទិន្នន័យទាំងអស់ជាទម្រង់ JSON
        [HttpGet]
        public IActionResult GetAllCategories()
        {
            // .ToList() នឹងបញ្ជាទៅ Database ឱ្យ Select ទិន្នន័យទាំងអស់ពី Table
            var categories = _context.Categories.ToList();
            return Ok(categories);
        }
    }
}

បញ្ជាក់ចំណុចប្រទាក់មុនបន្តទៅ View ៖

ដើម្បីឱ្យ API នេះមានទិន្នន័យពិតប្រាកដ អ្នកត្រូវរត់ពាក្យបញ្ជា dotnet ef migrations add Init និង dotnet ef database update នៅក្នុង Terminal ជាមុនសិន។ បន្ទាប់មក បើកកម្មវិធី SSMS រួចបញ្ចូលទិន្នន័យសាកល្បង (Insert Data) ចំនួន ២-៣ ជួរចូលក្នុង Table Categories ដោយផ្ទាល់ដៃ។ ចុងក្រោយ សូម Run Project ទុកចោល (API ត្រូវតែរត់ជានិច្ច)។

ដំណោះស្រាយ៖ ផ្នែកទី ២ (បញ្ចប់ MVC View)
៦️⃣ ជំហានទី ៦៖ តភ្ជាប់ API មកកាន់ View (JS Section)
សូមសរសេរកូដ <script> នេះ នៅខាងក្រោម Tag </table> នៅក្នុងឯកសារ index.html ខាងលើ។ កូដនេះនឹងធ្វើការខល (Fetch) ទៅ Backend យកទិន្នន័យ JSON មកតម្រៀបជាជួរ HTML ចូលក្នុង Table។
index.html (ផ្នែកកូដ JavaScript)
<script>
    // ទីតាំង URL របស់ Backend API (ត្រូវប្រាកដថា Port ត្រូវគ្នានឹងការ Run ពិតជាក់ស្តែង)
    const apiUrl = 'http://localhost:5000/api/categoryapi';

    // អនុវត្តមុខងារ Fetch ពេលឯកសារ HTML លោតចេញមកភ្លាម
    fetch(apiUrl)
        .then(response => response.json()) // បំប្លែងអត្ថបទទៅជា JSON Object
        .then(data => {
            const tableBody = document.getElementById('data-table');
            
            // Loop តាមចំនួនទិន្នន័យ Categories ដែលទទួលបានពី Database
            data.forEach(category => {
                const row = `
                    <tr>
                        <td>${category.id}</td>
                        <td>${category.name}</td>
                    </tr>
                `;
                // បញ្ចូលជួរថ្មីទៅក្នុងតារាង
                tableBody.innerHTML += row; 
            });
        })
        .catch(error => console.error('មានបញ្ហាក្នុងការទាញទិន្នន័យ៖', error));
</script>
